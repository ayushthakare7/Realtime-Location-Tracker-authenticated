import hmac, json, os
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import urlparse
from typing import TypedDict
from bson import ObjectId
from pymongo import MongoClient
from langchain_core.messages import SystemMessage, HumanMessage
from langchain_groq import ChatGroq
from langchain_huggingface import HuggingFaceEmbeddings
from langgraph.graph import StateGraph, START, END

class RevaState(TypedDict, total=False):
    question: str
    user_ids: list[str]
    context: dict
    evidence: list
    answer: str

PORT = int(os.getenv("RAG_PORT", "8001"))
INDEX = os.getenv("RAG_VECTOR_INDEX", "location_event_vector_index_384")
embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2",
                                   encode_kwargs={"normalize_embeddings": True})
mongo = MongoClient(os.environ["MONGO_URI"], serverSelectionTimeoutMS=5000)
db_name = os.getenv("MONGO_DB_NAME")
db = mongo[db_name] if db_name else mongo.get_default_database()
events = db[os.getenv("LOCATION_EVENTS_COLLECTION", "locationevents")]

def retrieve(state):
    ids = [ObjectId(value) for value in state["user_ids"]]
    vector = embeddings.embed_query(state["question"][:8000])
    pipeline = [{"$vectorSearch": {"index": INDEX, "path": "embedding", "queryVector": vector,
                 "numCandidates": 200, "limit": 12, "filter": {"userId": {"$in": ids}}}},
                {"$project": {"_id": 0, "username": 1, "eventType": 1, "placeName": 1,
                 "summary": 1, "startAt": 1, "endAt": 1, "durationMinutes": 1,
                 "recordedAt": 1}}]
    rows = list(events.aggregate(pipeline, maxTimeMS=8000))
    return {"evidence": rows}

def generate(state):
    llm = ChatGroq(model=os.getenv("GROQ_MODEL", "openai/gpt-oss-120b"),
                   api_key=os.environ["GROQ_API_KEY"], temperature=0.2, max_tokens=450)
    prompt = "Answer only from the authorized evidence. Never invent locations or times. If evidence is insufficient, say so. Evidence: " + json.dumps({"events": state.get("evidence", []), "context": state.get("context", {})}, default=str) + "\nQuestion: " + state["question"]
    result = llm.invoke([SystemMessage(content="You are Reva, a privacy-aware location history assistant."), HumanMessage(content=prompt)])
    return {"answer": str(result.content).strip()}

workflow = StateGraph(RevaState)
workflow.add_node("retrieve", retrieve)
workflow.add_node("generate", generate)
workflow.add_edge(START, "retrieve")
workflow.add_edge("retrieve", "generate")
workflow.add_edge("generate", END)
graph = workflow.compile()

class Handler(BaseHTTPRequestHandler):
    def send_json(self, status, payload):
        body = json.dumps(payload, default=str).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Cache-Control", "no-store")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)
    def do_GET(self):
        if urlparse(self.path).path != "/health":
            return self.send_json(404, {"success": False})
        try:
            mongo.admin.command("ping")
            return self.send_json(200, {"success": True, "service": "reva-rag"})
        except Exception:
            return self.send_json(503, {"success": False})
    def do_POST(self):
        if urlparse(self.path).path != "/ask":
            return self.send_json(404, {"success": False, "message": "Not found"})
        token = os.getenv("RAG_SERVICE_TOKEN", "")
        if not token or not hmac.compare_digest(self.headers.get("Authorization", ""), "Bearer " + token):
            return self.send_json(401, {"success": False, "message": "Unauthorized"})
        try:
            size = int(self.headers.get("Content-Length", "0"))
            if size < 1 or size > 65536:
                return self.send_json(413, {"success": False, "message": "Invalid request size"})
            payload = json.loads(self.rfile.read(size))
            question, ids = str(payload.get("question", "")).strip(), payload.get("userIds")
            context = payload.get("context", {})
            if not question or len(question) > 500 or not isinstance(ids, list) or len(ids) > 101 or not all(isinstance(x, str) for x in ids):
                return self.send_json(400, {"success": False, "message": "Invalid request"})
            result = graph.invoke({"question": question, "user_ids": ids, "context": context})
            return self.send_json(200, {"success": True, "answer": result.get("answer", "")})
        except Exception:
            return self.send_json(502, {"success": False, "message": "RAG workflow unavailable"})
    def log_message(self, fmt, *args):
        print("[RAG HTTP] " + (fmt % args), flush=True)

if __name__ == "__main__":
    ThreadingHTTPServer((os.getenv("RAG_HOST", "127.0.0.1"), PORT), Handler).serve_forever()
